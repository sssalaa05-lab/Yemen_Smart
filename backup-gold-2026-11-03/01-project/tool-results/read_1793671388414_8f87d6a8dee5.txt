   116→function fmtParams(n?: number): string {
   117→  if (!n) return '—'
   118→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B'
   119→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M'
   120→  return String(n)
   121→}
   122→function pct(a?: number): string {
   123→  return a != null ? (a * 100).toFixed(1) + '%' : '—'
   124→}

   153→function LossCurve({ curve, color = '#f59e0b' }: { curve: number[]; color?: string }) {
   154→  if (!curve || curve.length < 2) return null
   155→  const pts = curve.filter((_, i) => i % Math.max(1, Math.ceil(curve.length / 40)) === 0 || i === curve.length - 1)
   156→  const max = Math.max(...pts), min = Math.min(...pts)
   157→  const rng = max - min || 1
   158→  const coords = pts.map((v, i) => `${(i / (pts.length - 1)) * 100},${28 - ((v - min) / rng) * 24}`).join(' ')
   159→  return (
   160→    <div>
   161→      <div className="mb-2 flex items-center gap-2 text-sm text-neutral-400">
   162→        <TrendingUp className="h-4 w-4" style={{ color }} /> منحنى خسارة التدريب
   163→      </div>
   164→      <svg viewBox="0 0 100 30" preserveAspectRatio="none" className="h-24 w-full rounded-md bg-neutral-900/60 ring-1 ring-amber-500/10" role="img" aria-label="منحنى الخسارة">
   165→        <polyline points={coords} fill="none" stroke={color} strokeWidth="1.2" vectorEffect="non-scaling-stroke" />
   166→      </svg>
   167→      <div className="num mt-1 flex justify-between text-xs text-neutral-500" dir="ltr">
   168→        <span>{pts[0].toFixed(3)}</span>
   169→        <span>{pts[pts.length - 1].toFixed(3)}</span>
   170→      </div>
   171→    </div>
   172→  )
   173→}
