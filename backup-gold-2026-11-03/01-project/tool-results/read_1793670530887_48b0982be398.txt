    96→function fmtParams(n?: number): string {
    97→  if (!n) return '—'
    98→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B'
    99→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M'
   100→  return String(n)
   101→}
   102→function pct(a?: number): string {
   103→  return a != null ? (a * 100).toFixed(1) + '%' : '—'

   127→function PriceSpark({ series }: { series: number[] }) {
   128→  if (!series || series.length < 2) return null
   129→  const pts = series.filter((_, i) => i % Math.max(1, Math.ceil(series.length / 50)) === 0 || i === series.length - 1)
   130→  const max = Math.max(...pts), min = Math.min(...pts)
   131→  const rng = max - min || 1
   132→  const coords = pts.map((v, i) => `${(i / (pts.length - 1)) * 100},${28 - ((v - min) / rng) * 24}`).join(' ')
   133→  const up = pts[pts.length - 1] >= pts[0]
   134→  return (
   135→    <svg viewBox="0 0 100 30" preserveAspectRatio="none" className="h-24 w-full rounded-md bg-neutral-50" role="img" aria-label="سعر الذهب">
   136→      <polyline points={coords} fill="none" stroke={up ? '#d97706' : '#dc2626'} strokeWidth="1.2" vectorEffect="non-scaling-stroke" />
   137→    </svg>
   138→  )
   139→}

   141→function LossCurve({ curve }: { curve: number[] }) {
   142→  if (!curve || curve.length < 2) return null
   143→  const pts = curve.filter((_, i) => i % Math.max(1, Math.ceil(curve.length / 40)) === 0 || i === curve.length - 1)
   144→  const max = Math.max(...pts), min = Math.min(...pts)
   145→  const rng = max - min || 1
   146→  const coords = pts.map((v, i) => `${(i / (pts.length - 1)) * 100},${28 - ((v - min) / rng) * 24}`).join(' ')
   147→  return (
   148→    <div>
   149→      <div className="mb-2 flex items-center gap-2 text-sm text-neutral-600">
   150→        <TrendingUp className="h-4 w-4" /> منحنى خسارة التدريب
   151→      </div>
   152→      <svg viewBox="0 0 100 30" preserveAspectRatio="none" className="h-24 w-full rounded-md bg-neutral-50" role="img" aria-label="منحنى الخسارة">
   153→        <polyline points={coords} fill="none" stroke="#059669" strokeWidth="1.2" vectorEffect="non-scaling-stroke" />
   154→      </svg>
   155→      <div className="mt-1 flex justify-between text-xs text-neutral-400">
   156→        <span>البداية: {pts[0].toFixed(3)}</span>
   157→        <span>النهاية: {pts[pts.length - 1].toFixed(3)}</span>
   158→      </div>
   159→    </div>
   160→  )
   161→}
