   121→function fmtParams(n?: number): string {
   122→  if (!n) return '—'
   123→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B'
   124→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M'
   125→  return String(n)
   126→}
   127→function pct(a?: number): string {
   128→  return a != null ? (a * 100).toFixed(1) + '%' : '—'
   129→}

   158→function LossCurve({ curve, color = '#f59e0b' }: { curve: number[]; color?: string }) {
   159→  if (!curve || curve.length < 2) return null
   160→  const pts = curve.filter((_, i) => i % Math.max(1, Math.ceil(curve.length / 40)) === 0 || i === curve.length - 1)
   161→  const max = Math.max(...pts), min = Math.min(...pts)
   162→  const rng = max - min || 1
   163→  const coords = pts.map((v, i) => `${(i / (pts.length - 1)) * 100},${28 - ((v - min) / rng) * 24}`).join(' ')
   164→  return (
   165→    <div>
   166→      <div className="mb-2 flex items-center gap-2 text-sm text-neutral-400">
   167→        <TrendingUp className="h-4 w-4" style={{ color }} /> منحنى خسارة التدريب
   168→      </div>
   169→      <svg viewBox="0 0 100 30" preserveAspectRatio="none" className="h-24 w-full rounded-md bg-neutral-900/60 ring-1 ring-amber-500/10" role="img" aria-label="منحنى الخسارة">
   170→        <polyline points={coords} fill="none" stroke={color} strokeWidth="1.2" vectorEffect="non-scaling-stroke" />
   171→      </svg>
   172→      <div className="num mt-1 flex justify-between text-xs text-neutral-500" dir="ltr">
   173→        <span>{pts[0].toFixed(3)}</span>
   174→        <span>{pts[pts.length - 1].toFixed(3)}</span>
   175→      </div>
   176→    </div>
   177→  )
   178→}
